//go:build windows

// exitdiag measures, on Windows, whether GetExitCodeProcess can report a child's exit code before
// the child's handles are closed and before its process object is signaled.
package main

import (
	"bufio"
	"flag"
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"strconv"
	"syscall"
	"time"
)

const stillActive = 259
const synchronize = 0x00100000

func child(dir string, n, mb int) {
	var keep []*os.File
	for i := 0; i < n; i++ {
		f, err := os.OpenFile(filepath.Join(dir, "f"+strconv.Itoa(i)), os.O_CREATE|os.O_RDWR, 0o600)
		if err != nil {
			fmt.Println("openerr", err)
			os.Exit(3)
		}
		keep = append(keep, f)
	}
	mem := make([]byte, mb<<20)
	for i := 0; i < len(mem); i += 4096 {
		mem[i] = 1
	}
	fmt.Println("ready", len(keep), len(mem))
	_, _ = io.Copy(io.Discard, os.Stdin)
	os.Exit(0)
}

func main() {
	mode := flag.String("mode", "parent", "")
	dir := flag.String("dir", "", "")
	n := flag.Int("n", 2000, "files")
	mb := flag.Int("mb", 256, "MiB")
	iters := flag.Int("iters", 50, "")
	fresh := flag.Bool("fresh", false, "probe with a fresh OpenProcess per poll, as testutil.ProcessAlive does")
	flag.Parse()
	if *mode == "child" {
		child(*dir, *n, *mb)
		return
	}
	self, _ := os.Executable()
	codeBeforeSignal, codeBeforeClose, heldAtSignal, iterations := 0, 0, 0, 0
	var maxSig, maxClose time.Duration
	for it := 0; it < *iters; it++ {
		d, _ := os.MkdirTemp("", "exitdiag-")
		cmd := exec.Command(self, "-mode=child", "-dir="+d, "-n="+strconv.Itoa(*n), "-mb="+strconv.Itoa(*mb))
		in, _ := cmd.StdinPipe()
		out, _ := cmd.StdoutPipe()
		if err := cmd.Start(); err != nil {
			panic(err)
		}
		line, _ := bufio.NewReader(out).ReadString('\n')
		if len(line) < 5 || line[:5] != "ready" {
			panic("child: " + line)
		}
		pid := uint32(cmd.Process.Pid)
		h, err := syscall.OpenProcess(syscall.PROCESS_QUERY_INFORMATION|synchronize, false, pid)
		if err != nil {
			panic(err)
		}
		exitCode := func() uint32 {
			hh := h
			if *fresh {
				x, err := syscall.OpenProcess(syscall.PROCESS_QUERY_INFORMATION, false, pid)
				if err != nil {
					return 0 // ERROR_INVALID_PARAMETER: gone
				}
				defer syscall.CloseHandle(x)
				hh = x
			}
			var c uint32
			if err := syscall.GetExitCodeProcess(hh, &c); err != nil {
				panic(err)
			}
			return c
		}
		held := filepath.Join(d, "f0")
		_ = in.Close()
		for exitCode() == stillActive {
		}
		t0 := time.Now()
		sig, _ := syscall.WaitForSingleObject(h, 0)
		rmErr := os.Remove(held)
		if sig != 0 {
			codeBeforeSignal++
		}
		if rmErr != nil {
			codeBeforeClose++
		}
		var sigAt, closeAt time.Duration
		for sig != 0 {
			sig, _ = syscall.WaitForSingleObject(h, 0)
		}
		sigAt = time.Since(t0)
		if rmErr != nil {
			if rmErr = os.Remove(held); rmErr != nil {
				heldAtSignal++
			}
		}
		for rmErr != nil {
			rmErr = os.Remove(held)
			closeAt = time.Since(t0)
		}
		if sigAt > maxSig {
			maxSig = sigAt
		}
		if closeAt > maxClose {
			maxClose = closeAt
		}
		_ = cmd.Wait()
		_ = syscall.CloseHandle(h)
		_ = os.RemoveAll(d)
		iterations++
	}
	fmt.Printf("iterations=%d fresh=%v files=%d mb=%d exitCodeSetButNotSignaled=%d exitCodeSetButFileStillHeld=%d fileStillHeldOnceSignaled=%d maxSignalLag=%s maxFileHeldAfterSignalLag=%s\n",
		iterations, *fresh, *n, *mb, codeBeforeSignal, codeBeforeClose, heldAtSignal, maxSig, maxClose)
}
