type zzTimedHung struct {
	mu      sync.Mutex
	lastEnd time.Time
}

func (h *zzTimedHung) probe(_ ipc.Addr, by time.Time) bool {
	w := time.NewTimer(time.Until(by))
	<-w.C
	h.mu.Lock()
	h.lastEnd = time.Now()
	h.mu.Unlock()
	return false
}

func zzPct(xs []time.Duration, p float64) time.Duration {
	if len(xs) == 0 {
		return 0
	}
	s := append([]time.Duration(nil), xs...)
	sort.Slice(s, func(i, j int) bool { return s[i] < s[j] })
	i := int(p * float64(len(s)-1))
	return s[i]
}
