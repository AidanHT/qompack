"""TEMPORARY (w6-borrow): aggregate interleaved cold-start diagnostic rounds, before vs after.

Usage: python aggregate.py <log> ...   where each log's name contains "-before-" or "-after-" and a
round tag "rN". Prints, per tree, all rows pooled: spooled rows per source (and per source x mode),
multi-daemon and NOUP rows, and the session-start process wall time (the hook as its host times it)
p50/p90/p99/max; then the same per round, so a reader can see the comparison held round by round.
"""
import re
import sys
from collections import defaultdict


def pct(xs, q):
    if not xs:
        return 0
    s = sorted(xs)
    return s[int((len(s) - 1) * q)]


def ms(v):
    return int(re.sub(r'ms.*', '', v))


def rows_of(path):
    rows = []
    for l in open(path, encoding='utf-8', errors='replace').read().splitlines():
        if ' ROW mode=' not in l:
            continue
        d = {}
        for k, v in re.findall(r'(\w[\w-]*)=\s*([^\s]+)', l):
            d.setdefault(k, v)
        rows.append(d)
    return rows


def tree_of(path):
    return 'before' if '-before-' in path else 'after' if '-after-' in path else '?'


def round_of(path):
    m = re.search(r'-r(\d+)-', path)
    return int(m.group(1)) if m else 0


def describe(label, rows):
    walls = [ms(r['sswall']) for r in rows]
    sp = sum(r['spooled'] == 'true' for r in rows)
    print(f"{label}: rows={len(rows)} spooled={sp} ({100.0 * sp / max(len(rows), 1):.1f}%) "
          f"multi-daemon={sum(int(r['daemons']) > 1 for r in rows)} "
          f"noup={sum('ok=false' in r.get('up', '') for r in rows)} "
          f"sswall p50={pct(walls, .5)} p90={pct(walls, .9)} p99={pct(walls, .99)} "
          f"max={max(walls) if walls else 0} ms; over 13.5 s={sum(w > 13500 for w in walls)}; "
          f"over 15 s={sum(w > 15000 for w in walls)}")
    over = [ms(r['overhead']) for r in rows if 'overhead' in r]
    print(f"   outside doHook (process start + exit, the hookExitReserve share): p50={pct(over, .5)} "
          f"p99={pct(over, .99)} max={max(over) if over else 0} ms; over 1.5 s={sum(o > 1500 for o in over)}")
    for src in ('startup', 'resume', 'compact'):
        g = [r for r in rows if r['source'] == src]
        if not g:
            continue
        w = [ms(r['sswall']) for r in g]
        by_mode = defaultdict(lambda: [0, 0])
        for r in g:
            by_mode[r['mode']][0] += 1
            by_mode[r['mode']][1] += r['spooled'] == 'true'
        modes = ' '.join(f"{m}={s}/{n}" for m, (n, s) in sorted(by_mode.items()))
        print(f"   {src:8s} spooled={sum(r['spooled'] == 'true' for r in g)}/{len(g)} "
              f"sswall p50={pct(w, .5)} p90={pct(w, .9)} max={max(w)}   [{modes}]")


by_tree = defaultdict(list)
by_round = defaultdict(list)
for f in sys.argv[1:]:
    rs = rows_of(f)
    by_tree[tree_of(f)] += rs
    by_round[(round_of(f), tree_of(f))] += rs
for t in ('before', 'after'):
    if by_tree[t]:
        describe(f"== {t} (all rounds)", by_tree[t])
print()
for (rnd, t) in sorted(by_round):
    describe(f"-- round {rnd} {t}", by_round[(rnd, t)])
