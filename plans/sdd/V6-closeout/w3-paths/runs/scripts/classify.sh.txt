# red row ALONE at HEAD and at the base on Linux, then the Windows hot-path row at base and HEAD.
W=C:/Users/Quant/AppData/Local/Temp/claude/C--Users-Quant-Documents-Programming-Projects-qompack/9c57653d-e5ff-4791-8ab4-ad49457994c1/scratchpad/resume/paths-work

	echo "== $t start: $*" >> "$LOG"
	sh "$G" --prefix cx-w3-paths --out "$O" "$@" >> "$LOG" 2>&1

L=$WT/plans/sdd/V6-closeout/w3-paths/runs/17-hotpath-row-alone.log
{
	echo "# test/integration TestIntegration_HotPathWarmWithRealResidentState failed in the earlier seat's runs/19 (B-A/B-B"
	echo "# uncertifiable: 593 of 2130 requests deferred to the spool under co-load). Re-run ALONE: base (git archive 6aff949)"
	echo "# then HEAD $HEADREV, Windows, after this lane's Linux runs finished; other workstreams still load the machine."
	echo "# QOMPACK_UNDER_COLOAD not set."

go test ./test/integration -run '^TestIntegration_HotPathWarmWithRealResidentState$' -count=1 -timeout=30m -v >> "$L" 2>&1
echo "exit=$?" >> "$L"
echo "DONE" >> "$L"
echo "ALL-DONE $(date -u +%H:%M:%SZ)" >> "$LOG"
