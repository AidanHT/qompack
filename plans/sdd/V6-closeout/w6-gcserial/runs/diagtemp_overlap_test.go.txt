package store

// TEMPORARY DIAGNOSTIC (w6-gcserial), never committed: shows what two overlapping passes do to a
// resume cursor on the unfixed code.

import (
	"context"
	"fmt"
	"os"
	"path/filepath"
	"sync"
	"sync/atomic"
	"testing"
	"time"

	"github.com/stretchr/testify/require"

	"github.com/qompack/qompack/internal/paths"
)

func diagDiskBytes(t *testing.T, tp *testProject) (int64, int) {
	var total int64
	n := 0
	for _, rel := range tp.objectPaths(t) {
		info, err := os.Stat(paths.Long(filepath.Join(paths.Of(tp.Root).Objects, filepath.FromSlash(rel))))
		require.NoError(t, err)
		total += info.Size()
		n++
	}
	return total, n
}

func TestDiagTemp_OverlapDoubleResume(t *testing.T) {
	tp := newTestStore(t)
	ctx := context.Background()
	var refs []string
	for i := 0; i < 1200; i++ {
		r := gcSeed(t, tp, fmt.Sprintf("src/d%04d.ts", i), fmt.Sprintf("diag body %d, unique\n", i))
		if i%12 == 0 {
			refs = append(refs, r.Hash.String())
		}
	}
	writeCheckpointJSON(t, tp, "0001.json", refs...)
	st0, _ := tp.Store.Stats(ctx)
	disk0, n0 := diagDiskBytes(t, tp)
	t.Logf("before: Stats.Bytes=%d disk=%d objects=%d", st0.Bytes, disk0, n0)

	trunc, err := tp.Store.GC(ctx, GCPolicy{RetainDays: -1, RetainSessions: -1, Deadline: time.Nanosecond})
	require.NoError(t, err)
	t.Logf("truncated pass: truncated=%v deleted=%d freed=%d", trunc.Truncated, trunc.DeletedObjects, trunc.BytesFreed)
	require.FileExists(t, filepath.Join(paths.Of(tp.Root).State, gcStateFile))

	var arrived atomic.Int32
	both := make(chan struct{})
	var once sync.Once
	hook := func() {
		if arrived.Add(1) == 2 {
			once.Do(func() { close(both) })
		}
		select {
		case <-both:
		case <-time.After(5 * time.Second):
		}
	}
	prev := gcAfterHarvest.Swap(&hook)
	defer gcAfterHarvest.Store(prev)

	a := gcAsync(ctx, tp.Store, forceCollect)
	b := gcAsync(ctx, tp.Store, forceCollect)
	ra, rb := awaitGC(t, a), awaitGC(t, b)
	require.NoError(t, ra.err)
	require.NoError(t, rb.err)
	st1, _ := tp.Store.Stats(ctx)
	disk1, n1 := diagDiskBytes(t, tp)
	t.Logf("pass A: deleted=%d freed=%d scanned=%d", ra.rep.DeletedObjects, ra.rep.BytesFreed, ra.rep.ScannedObjects)
	t.Logf("pass B: deleted=%d freed=%d scanned=%d", rb.rep.DeletedObjects, rb.rep.BytesFreed, rb.rep.ScannedObjects)
	t.Logf("after: Stats.Bytes=%d disk=%d objects=%d (actually removed %d objects, %d bytes)", st1.Bytes, disk1, n1, n0-n1, disk0-disk1)
	t.Logf("reported deleted total=%d across truncated+A+B", trunc.DeletedObjects+ra.rep.DeletedObjects+rb.rep.DeletedObjects)
	t.Logf("accounting drift: Stats.Bytes - disk = %d", st1.Bytes-disk1)
}
