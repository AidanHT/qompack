+func zzProcs(bin string) string {
+	var b strings.Builder
+	entries, _ := os.ReadDir("/proc")
+	for _, e := range entries {
+		if _, err := strconv.Atoi(e.Name()); err != nil {
+			continue
+		}
+		cmd, err := os.ReadFile(filepath.Join("/proc", e.Name(), "cmdline"))
+		if err != nil || !bytes.Contains(cmd, []byte(bin)) {
+			continue
+		}
+		stat, _ := os.ReadFile(filepath.Join("/proc", e.Name(), "stat"))
+		fields := strings.Fields(string(stat))
+		state := ""
+		if len(fields) > 2 {
+			state = fields[2]
+		}
+		fmt.Fprintf(&b, "pid=%s state=%s cmd=%q ", e.Name(), state, string(bytes.ReplaceAll(cmd, []byte{0}, []byte{32})))
+	}
+	return b.String()
+}
