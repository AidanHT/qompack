+func zzSpool(root string) string {
+	var b strings.Builder
+	entries, _ := os.ReadDir(paths.Long(paths.Of(root).Spool))
+	for _, e := range entries {
+		if e.IsDir() {
+			continue
+		}
+		data, _ := os.ReadFile(paths.Long(filepath.Join(paths.Of(root).Spool, e.Name())))
+		fmt.Fprintf(&b, "%s(%dB,%dL) ", e.Name(), len(data), bytes.Count(data, []byte{10}))
+	}
+	if st, err := os.ReadFile(paths.Long(filepath.Join(paths.Of(root).State, "drain.json"))); err == nil {
+		fmt.Fprintf(&b, "drain.json=%s", strings.TrimSpace(string(st)))
+	}
+	return b.String()

+	}
