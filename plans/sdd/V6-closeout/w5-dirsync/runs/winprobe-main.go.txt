// Temporary diagnostic (w5-dirsync item 4): what Go's File.Sync does on a Windows directory handle.
package main

import (
	"fmt"
	"os"
	"runtime"
	"syscall"
)

func main() {
	fmt.Println("goos", runtime.GOOS, runtime.Version())
	dir, err := os.MkdirTemp("", "dirsync-probe-")
	if err != nil {
		panic(err)
	}
	defer os.RemoveAll(dir)

	f, err := os.Open(dir)
	fmt.Printf("os.Open(dir): err=%v\n", err)
	if err == nil {
		serr := f.Sync()
		var errno syscall.Errno
		fmt.Printf("  (*File).Sync on read-only dir handle: err=%v\n", serr)
		if e, ok := serr.(*os.PathError); ok {
			if n, ok := e.Err.(syscall.Errno); ok {
				errno = n
				fmt.Printf("  errno=%d\n", uintptr(errno))
			}
		}
		f.Close()
	}
	g, err := os.OpenFile(dir, os.O_RDWR, 0)
	fmt.Printf("os.OpenFile(dir, O_RDWR): err=%v\n", err)
	if err == nil {
		fmt.Printf("  (*File).Sync on O_RDWR dir handle: err=%v\n", g.Sync())
		g.Close()
	}
	// Raw CreateFile with GENERIC_WRITE + FILE_FLAG_BACKUP_SEMANTICS, then FlushFileBuffers.
	p, _ := syscall.UTF16PtrFromString(dir)
	h, err := syscall.CreateFile(p, syscall.GENERIC_READ|syscall.GENERIC_WRITE,
		syscall.FILE_SHARE_READ|syscall.FILE_SHARE_WRITE|syscall.FILE_SHARE_DELETE, nil,
		syscall.OPEN_EXISTING, syscall.FILE_FLAG_BACKUP_SEMANTICS, 0)
	fmt.Printf("CreateFile(dir, GENERIC_WRITE, BACKUP_SEMANTICS): err=%v\n", err)
	if err == nil {
		fmt.Printf("  FlushFileBuffers(dir handle): err=%v\n", syscall.FlushFileBuffers(h))
		syscall.CloseHandle(h)
	}
}
