"""TEMPORARY: instrument a scratch export of the tree with zzdiag marks (w5-coldstart diagnostic).

Usage: python apply.py <tree-root>. Each edit is anchored on exact text; an anchor that is missing
is reported and skipped so the same script serves the before and after trees.
"""
import os
import shutil
import sys

root = sys.argv[1]
here = os.path.dirname(os.path.abspath(__file__))
os.makedirs(os.path.join(root, "internal", "zzdiag"), exist_ok=True)
shutil.copy(os.path.join(here, "zzdiag.go"), os.path.join(root, "internal", "zzdiag", "zzdiag.go"))

IMPORT = '\t"github.com/qompack/qompack/internal/zzdiag"\n'


def edit(rel, pairs):
    p = os.path.join(root, rel)
    s = open(p, encoding="utf-8").read()
    if IMPORT not in s:
        s = s.replace("import (\n", "import (\n" + IMPORT, 1)
    for mode, anchor, text in pairs:
        if anchor not in s:
            print(f"MISSING anchor in {rel}: {anchor[:70]!r}")
            continue
        if mode == "after":
            s = s.replace(anchor, anchor + text, 1)
        else:
            s = s.replace(anchor, text + anchor, 1)
    open(p, "w", encoding="utf-8").write(s)
